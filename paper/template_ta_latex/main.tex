% Menerima argumen opsional untuk penamaan di Daftar Isi jika judul halaman panjang.
\newcommand{\halamanAwal}[2][]{%
  \clearpage
  \thispagestyle{plain}
  \phantomsection
  \if\relax\detokenize{#1}\relax
    \addcontentsline{toc}{chapter}{#2}%
  \else
    \addcontentsline{toc}{chapter}{#1}%
  \fi
  \begin{center}
    {\bfseries\normalsize #2\par}
  \end{center}
  \vspace{0.8cm}
}

\makeatletter
\newcommand{\halamanDaftar}[2]{%
  \clearpage
  \thispagestyle{plain}
  \phantomsection
  \addcontentsline{toc}{chapter}{#1}%
  \begin{center}
    {\bfseries\normalsize #1\par}
  \end{center}
  \vspace{0.8cm}
  \begingroup
    \parindent\z@
    \parskip\cftparskip
    \@starttoc{#2}%
  \endgroup
}
\makeatother
